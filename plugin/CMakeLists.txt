cmake_minimum_required(VERSION 3.22)
project(LoopForge VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
if(LOOPFORGE_JUCE_PATH)
 add_subdirectory(${LOOPFORGE_JUCE_PATH} JUCE)
else()
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG d6181bde38d858c283c3b7bf699ce6340c050b5d)
 FetchContent_MakeAvailable(JUCE)
endif()
juce_add_plugin(LoopForge COMPANY_NAME LoopForge PLUGIN_MANUFACTURER_CODE LpFg PLUGIN_CODE Lf02
 FORMATS VST3 Standalone PRODUCT_NAME "LoopForge AI MIDI"
 IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT TRUE IS_MIDI_EFFECT FALSE
 COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(LoopForge PRIVATE Source/Plugin.cpp)
target_compile_definitions(LoopForge PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(LoopForge PRIVATE juce::juce_audio_utils juce::juce_audio_processors juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
enable_testing()
add_executable(scheduler_test tests/SchedulerTest.cpp)
add_test(NAME scheduler COMMAND scheduler_test)
add_executable(processor_test tests/ProcessorTest.cpp)
target_include_directories(processor_test PRIVATE $<TARGET_PROPERTY:LoopForge,INCLUDE_DIRECTORIES>)
target_compile_definitions(processor_test PRIVATE $<TARGET_PROPERTY:LoopForge,COMPILE_DEFINITIONS>)
target_link_libraries(processor_test PRIVATE LoopForge)
add_test(NAME processor COMMAND processor_test ${CMAKE_CURRENT_SOURCE_DIR}/tests/idea.json)
add_executable(stress_test tests/StressTest.cpp)
target_include_directories(stress_test PRIVATE $<TARGET_PROPERTY:LoopForge,INCLUDE_DIRECTORIES>)
target_compile_definitions(stress_test PRIVATE $<TARGET_PROPERTY:LoopForge,COMPILE_DEFINITIONS>)
target_link_libraries(stress_test PRIVATE LoopForge)
add_test(NAME native_stress COMMAND stress_test)
